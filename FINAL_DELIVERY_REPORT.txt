🚀 USAGE EXAMPLE
═══════════════════════════════════════════════════════════════════════

// workflow.nf
process myProcess {
    executor 'codespaces'
    
    output:
    path 'result.txt'
    
    script:
    """
    echo "Hello from Codespaces!" > result.txt
    """
}

// nextflow.config
process {
    executor = 'codespaces'
}

// Execute
$ nextflow run workflow.nf
